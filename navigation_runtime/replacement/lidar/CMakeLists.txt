cmake_minimum_required(VERSION 3.16)
project(feedbot_ltr2_native)
set(CMAKE_CXX_STANDARD 17)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
add_executable(ltr2_replay replay.cpp)
add_executable(ltr2_node node.cpp)
ament_target_dependencies(ltr2_node rclcpp sensor_msgs)
add_executable(ltr2_lifecycle_test lifecycle_test.cpp)
enable_testing()
add_test(NAME lifecycle COMMAND ltr2_lifecycle_test)
set(LDCP_SOURCE "" CACHE PATH "Pinned vendor SDK source for control-plane fixture")
if(LDCP_SOURCE)
  add_subdirectory(${LDCP_SOURCE} ${CMAKE_BINARY_DIR}/ldcp)
  add_executable(ltr2_sdk_fixture sdk_fixture.cpp)
  target_link_libraries(ltr2_sdk_fixture ldcp_sdk)
  find_package(rclcpp_lifecycle REQUIRED)
  find_package(std_msgs REQUIRED)
  add_executable(ltr2_driver managed_node.cpp)
  target_link_libraries(ltr2_driver ldcp_sdk)
  ament_target_dependencies(ltr2_driver rclcpp rclcpp_lifecycle sensor_msgs std_msgs)
endif()
