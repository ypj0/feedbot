cmake_minimum_required(VERSION 3.16)
project(feedbot_chassis_ros2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(hardware_interface REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
find_package(diagnostic_msgs REQUIRED)
find_package(realtime_tools REQUIRED)
set(CHASSIS_CORE "${CMAKE_CURRENT_SOURCE_DIR}/../../../navigation_runtime/replacement/chassis" CACHE PATH "Frozen chassis headers")
add_library(feedbot_chassis_system SHARED src/system.cpp)
target_include_directories(feedbot_chassis_system PRIVATE include ${CHASSIS_CORE})
ament_target_dependencies(feedbot_chassis_system hardware_interface pluginlib rclcpp diagnostic_msgs realtime_tools)
pluginlib_export_plugin_description_file(hardware_interface plugin.xml)
enable_testing()
add_executable(chassis_session_test tests/session_test.cpp)
target_include_directories(chassis_session_test PRIVATE include ${CHASSIS_CORE})
target_link_libraries(chassis_session_test pthread util)
add_test(NAME chassis_session COMMAND chassis_session_test)
add_executable(chassis_tx_contract_test tests/tx_contract_test.cpp)
target_include_directories(chassis_tx_contract_test PRIVATE include ${CHASSIS_CORE})
target_link_libraries(chassis_tx_contract_test ${CMAKE_DL_LIBS})
add_test(NAME chassis_tx_contract COMMAND chassis_tx_contract_test)
add_executable(chassis_release_connection_test tests/release_connection_test.cpp)
target_include_directories(chassis_release_connection_test PRIVATE include ${CHASSIS_CORE})
target_link_libraries(chassis_release_connection_test pthread util)
add_test(NAME chassis_release_connection COMMAND chassis_release_connection_test)
foreach(check test odometry_test)
  add_executable(frozen_${check} ${CHASSIS_CORE}/${check}.cpp)
  add_test(NAME frozen_${check} COMMAND frozen_${check})
endforeach()
install(TARGETS feedbot_chassis_system DESTINATION lib)
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
ament_package()
