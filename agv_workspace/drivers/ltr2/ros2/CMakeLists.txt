cmake_minimum_required(VERSION 3.16)
project(feedbot_ltr2_ros2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_lifecycle REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(diagnostic_msgs REQUIRED)
find_package(realtime_tools REQUIRED)
set(LDCP_SOURCE "" CACHE PATH "Pinned vendor LDCP SDK source")
if(NOT LDCP_SOURCE)
  message(FATAL_ERROR "LDCP_SOURCE must reference the validated SDK")
endif()
add_subdirectory(${LDCP_SOURCE} ${CMAKE_BINARY_DIR}/ldcp)
include_directories(../include include)
add_executable(ltr2_ros2 src/node.cpp)
ament_target_dependencies(ltr2_ros2 rclcpp rclcpp_lifecycle sensor_msgs diagnostic_msgs realtime_tools)
target_link_libraries(ltr2_ros2 ldcp_sdk)
enable_testing()
add_executable(ltr2_transport_fixture tests/transport_fixture.cpp)
ament_target_dependencies(ltr2_transport_fixture rclcpp rclcpp_lifecycle sensor_msgs)
add_executable(ltr2_message_test tests/test_message.cpp)
ament_target_dependencies(ltr2_message_test rclcpp sensor_msgs)
add_test(NAME message_conversion COMMAND ltr2_message_test)
add_executable(ltr2_handoff_test tests/test_handoff.cpp)
target_compile_options(ltr2_handoff_test PRIVATE -UNDEBUG)
ament_target_dependencies(ltr2_handoff_test rclcpp sensor_msgs realtime_tools)
add_test(NAME scan_handoff COMMAND ltr2_handoff_test)
set_tests_properties(scan_handoff PROPERTIES TIMEOUT 10)
foreach(check core freshness lifecycle udp_input)
  add_executable(ltr2_${check}_test ../tests/test_${check}.cpp)
  add_test(NAME ${check} COMMAND ltr2_${check}_test)
endforeach()
install(TARGETS ltr2_ros2 DESTINATION lib/${PROJECT_NAME})
ament_package()
