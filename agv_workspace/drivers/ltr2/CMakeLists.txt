cmake_minimum_required(VERSION 3.10)
project(feedbot_ltr2_candidate LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
option(LTR2_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
option(LTR2_BUILD_ROS1 "Build bounded Linux ROS1 passive observer" OFF)
option(LTR2_BUILD_MANAGED "Build independent ROS1 control and UDP driver" OFF)
set(LDCP_SOURCE "" CACHE PATH "Pinned vendor LDCP SDK source")
if(LTR2_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  set(CMAKE_EXE_LINKER_FLAGS "${CMAKE_EXE_LINKER_FLAGS} -fsanitize=address,undefined")
endif()
include_directories(include)
enable_testing()
add_executable(ltr2_core_test tests/test_core.cpp)
add_test(NAME core COMMAND ltr2_core_test)
add_executable(ltr2_transport_test tests/test_transport.cpp)
add_test(NAME transport COMMAND ltr2_transport_test)
add_executable(ltr2_freshness_test tests/test_freshness.cpp)
add_test(NAME freshness COMMAND ltr2_freshness_test)
add_executable(ltr2_lifecycle_test tests/test_lifecycle.cpp)
add_test(NAME lifecycle COMMAND ltr2_lifecycle_test)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  add_executable(ltr2_passive_input_test tests/test_passive_input.cpp)
  add_executable(ltr2_udp_test tests/test_udp_input.cpp)
  add_test(NAME exclusive_udp COMMAND ltr2_udp_test)
endif()
add_executable(ltr2_replay src/replay.cpp)
if(LDCP_SOURCE)
  add_subdirectory(${LDCP_SOURCE} ${CMAKE_BINARY_DIR}/ldcp)
  add_executable(ltr2_control_fixture tests/control_fixture.cpp)
  target_link_libraries(ltr2_control_fixture ldcp_sdk)
  find_program(LTR2_PYTHON3 python3)
  if(LTR2_PYTHON3)
    add_test(NAME sdk_control COMMAND ${LTR2_PYTHON3}
      ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_sdk_control.py $<TARGET_FILE:ltr2_control_fixture>)
  endif()
endif()
if(LTR2_BUILD_ROS1 OR LTR2_BUILD_MANAGED)
  if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "Passive observer requires Linux AF_PACKET")
  endif()
  find_package(catkin REQUIRED COMPONENTS roscpp sensor_msgs diagnostic_msgs std_srvs)
endif()
if(LTR2_BUILD_ROS1)
  add_executable(ltr2_passive_ros1 src/passive_ros1.cpp)
  target_include_directories(ltr2_passive_ros1 PRIVATE ${catkin_INCLUDE_DIRS})
  target_link_libraries(ltr2_passive_ros1 ${catkin_LIBRARIES})
endif()
if(LTR2_BUILD_MANAGED)
  if(NOT LDCP_SOURCE)
    message(FATAL_ERROR "Managed driver requires pinned LDCP_SOURCE")
  endif()
  add_executable(ltr2_managed_ros1 src/managed_ros1.cpp)
  target_include_directories(ltr2_managed_ros1 PRIVATE ${catkin_INCLUDE_DIRS})
  target_link_libraries(ltr2_managed_ros1 ldcp_sdk ${catkin_LIBRARIES})
endif()
