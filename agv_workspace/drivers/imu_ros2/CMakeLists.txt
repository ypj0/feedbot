cmake_minimum_required(VERSION 3.16)
project(feedbot_imu_ros2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(diagnostic_msgs REQUIRED)
find_package(realtime_tools REQUIRED)
set(IMU_CORE "${CMAKE_CURRENT_SOURCE_DIR}/../../../navigation_runtime/replacement/imu" CACHE PATH "Frozen IMU decoder")
set(CHASSIS_CORE "${CMAKE_CURRENT_SOURCE_DIR}/../../../navigation_runtime/replacement/chassis" CACHE PATH "Frozen serial transport")
add_executable(imu_node src/node.cpp)
target_include_directories(imu_node PRIVATE include ${IMU_CORE} ${CHASSIS_CORE})
ament_target_dependencies(imu_node rclcpp sensor_msgs diagnostic_msgs realtime_tools)
enable_testing()
add_executable(imu_message_test tests/message_test.cpp)
target_include_directories(imu_message_test PRIVATE include ${IMU_CORE})
ament_target_dependencies(imu_message_test rclcpp sensor_msgs)
add_test(NAME imu_message COMMAND imu_message_test)
add_executable(callback_gap_histogram_test tests/callback_gap_histogram_test.cpp)
target_include_directories(callback_gap_histogram_test PRIVATE include)
target_compile_options(callback_gap_histogram_test PRIVATE -UNDEBUG)
add_test(NAME callback_gap_histogram COMMAND callback_gap_histogram_test)
add_executable(publish_handoff_test tests/publish_handoff_test.cpp)
target_compile_options(publish_handoff_test PRIVATE -UNDEBUG)
ament_target_dependencies(publish_handoff_test rclcpp sensor_msgs realtime_tools)
add_test(NAME publish_handoff COMMAND publish_handoff_test)
set_tests_properties(publish_handoff PROPERTIES TIMEOUT 10)
add_executable(latest_imu_publisher_test tests/latest_imu_publisher_test.cpp)
target_include_directories(latest_imu_publisher_test PRIVATE include)
target_compile_options(latest_imu_publisher_test PRIVATE -UNDEBUG)
ament_target_dependencies(latest_imu_publisher_test sensor_msgs)
target_link_libraries(latest_imu_publisher_test pthread)
add_test(NAME latest_imu_publisher COMMAND latest_imu_publisher_test)
set_tests_properties(latest_imu_publisher PROPERTIES TIMEOUT 10)
add_executable(frozen_imu_test ${IMU_CORE}/test.cpp)
add_test(NAME frozen_imu COMMAND frozen_imu_test)
install(TARGETS imu_node DESTINATION lib/${PROJECT_NAME})
ament_package()
